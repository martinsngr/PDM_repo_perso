FoamFile
{
    version     2.0;
    format      ascii;
    class       volScalarField;
    location    "0";
    object      Sb;
}
// * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * //

dimensions      [0 0 0 0 0 0 0];

// uniform dry state; ./run copies this file to 0/Sb and setFields adds the
// prewetted ramp at the bottom (system/setFieldsDict)
internalField   uniform 2e-3;

boundaryField
{
    // infinite water reservoir: the yarn end face at x-min is held at Sb 0.99
    // (no clamped cellZone), as ../1D_vertical_impes_inletRes
    left
    {
        type            fixedValue;
        value           uniform 0.99;
    }

    // no capillary flux through the top or the closed lateral surface (Ub is
    // fixed 0 there anyway); a fixed value crashes the 1D hybrid case
    "(right|yarn_to_fluid)"
    {
        type            zeroGradient;
    }
}
